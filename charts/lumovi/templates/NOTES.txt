kind: Config
clusters:
  - name: {{ .Values.clusterName }}
    cluster:
      server: $SERVER
      certificate-authority-data: {{ "{{" }} index .data \"ca.crt\" {{ "}}" }}
users:
  - name: {{ .Values.clusterName }}
    user:
      token: {{ "{{" }} .data.token | base64decode {{ "}}" }}
contexts:
  - name: {{ .Values.clusterName }}
    context:
      cluster: {{ .Values.clusterName }}
      user: {{ .Values.clusterName }}
"

Keep it secret: whoever has the token can act as anyone here.
